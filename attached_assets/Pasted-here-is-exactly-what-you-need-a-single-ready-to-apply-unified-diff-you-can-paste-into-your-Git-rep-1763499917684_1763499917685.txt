+async function loadBundleDetail(bundleId: number, user: AuthUser | undefined): Promise<BundleDetail> {
+  const [bundle] = await dbRead.select().from(bundles).where(eq(bundles.id, bundleId));
+
+  if (!bundle) {

+  }
+
+  ensureOwner(user, bundle.ownerId);

+  const rows = await dbRead.execute(
+    sql`
+      SELECT

+        j.id       AS "judgmentId",
+        j.debtor_name AS "debtorName",
+        j.case_number AS "caseNumber",
+        j.judgment_amount AS "judgmentAmount",
+        pe.recommended_purchase_price AS "recommendedPurchasePrice"
+      FROM bundle_judgments bj
+      JOIN judgments j ON j.id = bj.judgment_id
+      LEFT JOIN price_estimates pe
+        ON pe.judgment_id = j.id AND pe.scenario_name = 'base'
+      WHERE bj.bundle_id = ${bundleId}
+    `
+  );
+
+  const trancheKeys: TrancheKey[] = ['senior', 'mezz', 'equity'];
+
+  const trancheMetrics: Record<TrancheKey, BundleMetrics> = {
+    senior: { faceValue: 0, recommendedPurchasePrice: 0, judgmentCount: 0 },
+    mezz: { faceValue: 0, recommendedPurchasePrice: 0, judgmentCount: 0 },
+    equity: { faceValue: 0, recommendedPurchasePrice: 0, judgmentCount: 0 }
+  };
+
+  const items: BundleJudgmentItem[] = [];
+
+  for (const row of rows.rows as any[]) {
+    const tranche = (row.tranche ?? 'senior') as TrancheKey;
+    const face = Number(row.judgmentAmount ?? 0);
+    const price = Number(row.recommendedPurchasePrice ?? 0);
+
+    trancheMetrics[tranche].faceValue += face;
+    trancheMetrics[tranche].recommendedPurchasePrice += price;
+    trancheMetrics[tranche].judgmentCount += 1;
+
+    items.push({
+      id: Number(row.judgmentId),
+      debtorName: row.debtorName,
+      caseNumber: row.caseNumber,
+      judgmentAmount: face,
+      tranche,
+      recommendedPurchasePrice: price || undefined
+    });
+  }
+
+  const overall: BundleMetrics = trancheKeys.reduce(
+    (acc, key) => ({
+      faceValue: acc.faceValue + trancheMetrics[key].faceValue,
+      recommendedPurchasePrice:
+        acc.recommendedPurchasePrice + trancheMetrics[key].recommendedPurchasePrice,
+      judgmentCount: acc.judgmentCount + trancheMetrics[key].judgmentCount
+    }),
+    { faceValue: 0, recommendedPurchasePrice: 0, judgmentCount: 0 }
+  );
+
+  return {
+    id: bundle.id,
+    name: bundle.name,
+    description: bundle.description,
+    status: bundle.status,
+    metrics: {
+      overall,
+      tranches: trancheMetrics
+    },
+    judgments: items
+  };
