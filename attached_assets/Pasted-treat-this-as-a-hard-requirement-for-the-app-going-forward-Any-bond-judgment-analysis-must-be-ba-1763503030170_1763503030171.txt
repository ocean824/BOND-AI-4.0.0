// server/services/bondAnalyticsService.ts
import { bonds } from '../../db/schema.js';
import { dbRead } from '../db/client.js';
import { eq } from 'drizzle-orm';
import { NotFoundError, ValidationError } from '../errors/index.js';

export interface BondAnalytics {
  price: number;
  yieldToMaturity: number;
  accruedInterest: number;
  cleanPrice: number;
  dirtyPrice: number;
  macaulayDuration: number;
  modifiedDuration: number;
  convexity: number;
}

export async function analyzeBondByCusip(cusip: string): Promise<BondAnalytics> {
  const [bond] = await dbRead
    .select()
    .from(bonds)
    .where(eq(bonds.cusip, cusip));

  if (!bond) {
    throw new NotFoundError(`No bond found for CUSIP ${cusip}`);
  }

  if (!bond.lastPrice || !bond.couponRate || !bond.maturityDate || !bond.couponFrequency) {
    throw new ValidationError('Bond is missing required fields for analytics');
  }

  // Here we assume all fields are real values, not mock.
  const result = calculateBondAnalytics({
    price: Number(bond.lastPrice),
    couponRate: Number(bond.couponRate),
    couponFrequency: bond.couponFrequency,
    maturityDate: bond.maturityDate,
    faceValue: Number(bond.faceValue ?? 1000)
  });

  return result;
}

Where calculateBondAnalytics is pure math based on real price / coupon / maturity. It never generates fake inputs if something’s missing — it just throws a proper error.

import fetch from 'node-fetch';

export async function fetchRealBondSnapshot(cusip: string) {
  const res = await fetch(`${process.env.BOND_DATA_PROVIDER_URL}/bonds/${cusip}`, {
    headers: {
      Authorization: `Bearer ${process.env.BOND_DATA_API_KEY}`
    }
  });

  if (!res.ok) {
    throw new Error(`Provider error: ${res.status}`);
  }

  const data = await res.json();
  // No mock: just normalize provider’s real data.
  return {
    price: data.price,

    // ...
  };
}
