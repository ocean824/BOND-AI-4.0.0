2.1. Core auth + role helpers (Express-style)
// src/middleware/auth.ts
import type { Request, Response, NextFunction } from "express";
import type { AppRole, AuthUser } from "../types/auth";

declare global {
  namespace Express {
    interface Request {
      user?: AuthUser;
    }
  }
}

// Generic role gate
export function requireAuth(allowedRoles?: AppRole[]) {
  return (req: Request, res: Response, next: NextFunction) => {
    const user = req.user;
    if (!user) return res.status(401).json({ error: "Unauthorized" });

    if (allowedRoles && !allowedRoles.includes(user.role)) {
      return res.status(403).json({ error: "Forbidden" });
    }

    next();
  };
}

// Simple ADMIN-only wrapper
export function requireAdminOnly(req: Request, res: Response, next: NextFunction) {
  const user = req.user;
  if (!user || user.role !== "ADMIN") {
    return res.status(403).json({ error: "Admin access required" });
  }
  next();
}

// Global guard: INVESTORS can *only* see /auth/* + /investor/*
export function blockInvestorsForNonInvestorRoutes(
  req: Request,
  res: Response,
  next: NextFunction,
) {
  const user = req.user;

  if (user?.role === "INVESTOR") {
    const path = req.path;

    if (path.startsWith("/auth/") || path.startsWith("/investor/")) {
      return next();
    }

    return res.status(403).json({
      error: "Investors cannot access this resource",
    });
  }

  // all other roles (ADMIN/UNDERWRITER/INTERNAL) proceed normally
  next();
}

// src/middleware/suretyPin.ts
import type { Request, Response, NextFunction } from "express";

export function requireSuretyPin(req: Request, res: Response, next: NextFunction) {
  const user = req.user;
  if (!user || user.role !== "ADMIN") {
    return res.status(403).json({ error: "Admin access required" });
  }

  const headerPin = req.header("x-surety-pin");
  const expected = process.env.SURETY_ADMIN_PIN;

  if (!expected) {
    return res.status(500).json({ error: "Surety PIN not configured" });
  }

  if (headerPin !== expected) {
    return res.status(403).json({ error: "Invalid Surety PIN" });
  }

  next();
}

3.2. ADMIN-only route to generate docs for a step
// src/routes/adminBondDocs.ts
import { Router } from "express";
import { requireAdminOnly } from "../middleware/auth";
import { generateDocsForBondStep } from "../services/documentEngine";

export const adminBondDocsRouter = Router();

// 🔐 Only ADMIN can generate documents for any step
adminBondDocsRouter.post(
  "/admin/bonds/:bondId/generate-docs",
  requireAdminOnly,
  async (req, res) => {
    const { bondId } = req.params;
    const { step, suretyCaseId, trustId } = req.body as {
      step: string;
      suretyCaseId?: string;
      trustId?: string;
    };

    try {
      const docs = await generateDocsForBondStep({
        bondId,
        step: step as any,
        suretyCaseId,
        trustId,
      });

      return res.json({ docs });
    } catch (err: any) {
      console.error(err);
      return res.status(500).json({ error: "Doc generation failed" });
    }
  },
);

4. “Every step of the bond process” – document checklist

Here’s a clean matrix you (or a lawyer) can refine, but the app is architected to handle every row via the documentEngine above:

Step in pipeline	What happens in app	Example docs the app should generate
1. Judgment → Bond	Admin converts judgment into a Bond record	BOND_TERM_SHEET, BOND_INSTRUMENT (internal term sheet + the actual bond/IOU form)
2. Perfection	Bond passes internal compliance, liens/filings done	PERFECTION_CERT (certificate of perfection, internal memo that all filings are complete)
3. Surety underwriting	Surety Trust analyzes the judgment bond	UNDERWRITING_REPORT (summary of risk, capacity check, collateral/indemnity recommendations)
4. Surety activation	Trust agrees to act as surety	GIA (General Indemnity Agreement), COLLATERAL_AGREEMENT, SURETY_TRUST_RESOLUTION (board/trust resolution authorizing surety)
5. Exchange listing	Bond is listed for auction to investors	EXCHANGE_OFFERING_SHEET (summary for investors, no sensitive debtor details if you choose to mask)
6. Exchange sale / closing	Auction completes, winning bid + settlement	INVESTOR_SUBSCRIPTION (or purchase agreement), TRADE_CONFIRMATION / closing statement
7. Trust allocation & minutes	Proceeds allocated via Development Trust / management LLC	TRUST_BOARD_MINUTES, TRUST_ALLOCATION_RESOLUTION (cashflow split, tax reserves, project funding, etc.)

All of those doc types are centralized in code as DocType, and only ADMIN can trigger them.
