// middleware/auth.ts
import { Request, Response, NextFunction } from "express";

export type AppRole = "ADMIN" | "UNDERWRITER" | "INTERNAL" | "INVESTOR";

export interface AuthUser {
  id: string;
  role: AppRole;
  email: string;
}

declare global {
  namespace Express {
    interface Request {
      user?: AuthUser;
    }
  }
}

export function requireAuth(
  allowedRoles?: AppRole[]
) {
  return (req: Request, res: Response, next: NextFunction) => {
    const user = req.user;
    if (!user) return res.status(401).json({ error: "Unauthorized" });

    if (allowedRoles && !allowedRoles.includes(user.role)) {
      return res.status(403).json({ error: "Forbidden" });
    }
    next();
  };
}

// "Investor gate": investors CANNOT access non-investor routes
export function blockInvestorsForNonInvestorRoutes(
  req: Request,
  res: Response,
  next: NextFunction
) {
  const user = req.user;
  if (user?.role === "INVESTOR") {
    // Only allow /auth/* and /investor/*
    if (
      req.path.startsWith("/auth/") ||
      req.path.startsWith("/investor/")
    ) {
      return next();
    }
    return res.status(403).json({ error: "Investors cannot access this resource" });
  }
  next();
}

3. Backend Logic: Triggers, Auctions, Bidding, and Reserve
3.1. Automatic trigger: perfected bond → auction
Some other part of your app will mark bonds as perfected. When that happens – and the bond is approved for sale – we can auto-create an auction.

Event hook / service:

// services/bondPerfection.ts
import { db } from "../db";
import { bonds } from "../db/schema/bonds";
import { auctions } from "../db/schema/auctions";
import { eq } from "drizzle-orm";
import dayjs from "dayjs";

export async function handleBondPerfected(bondId: string, reservePrice: number) {
  const [bond] = await db.select().from(bonds).where(eq(bonds.id, bondId));
  if (!bond) throw new Error("Bond not found");

  const startsAt = new Date(); // or schedule/future start if needed
  const endsAt = dayjs(startsAt).add(5, "day").toDate();

  const [auction] = await db
    .insert(auctions)
    .values({
      bondId,
      reservePrice,
      status: "ACTIVE",
      startsAt,
      endsAt,
    })
    .returning();

  await db
    .update(bonds)
    .set({
      isPerfected: true,
      perfectedAt: new Date(),
      currentAuctionId: auction.id,
    })
    .where(eq(bonds.id, bondId));

  return auction;
}

        next30Days: 0,
        later: 0,

3.3. Place bid endpoint (with reserve logic)
Business rules:

Only INVESTOR role.

Only while auction.status === "ACTIVE" and now between startsAt and endsAt.

Bid must be > current highest bid (or a minimum increment you define).

If bid amount ≥ reservePrice:

Mark auction RESERVE_MET → then immediately COMPLETED (per your requirement: “taken down once reserve price has been successfully bid”).

Set winningBidId.

Remove from active listing.

// routes/investorBids.ts
import { Router } from "express";
import { requireAuth } from "../middleware/auth";
import { auctions } from "../db/schema/auctions";
import { auctionBids } from "../db/schema/auctionBids";
import { db } from "../db";
import { eq } from "drizzle-orm";

export const investorBidsRouter = Router();

investorBidsRouter.post(
  "/investor/auctions/:id/bids",
  requireAuth(["INVESTOR"]),
  async (req, res) => {
    const investor = req.user!;
    const { id } = req.params;
    const { amount } = req.body as { amount: number };

    if (!amount || amount <= 0) {
      return res.status(400).json({ error: "Bid amount must be positive" });
    }

    const [auction] = await db
      .select()
      .from(auctions)
      .where(eq(auctions.id, id));

    if (!auction) return res.status(404).json({ error: "Auction not found" });

    const now = new Date();
    if (auction.status !== "ACTIVE" || now < auction.startsAt || now > auction.endsAt) {
      return res.status(400).json({ error: "Auction is not open for bidding" });
    }

    // Get current highest bid
    const [highestBid] = await db
      .select()
      .from(auctionBids)
      .where(eq(auctionBids.auctionId, auction.id))
      .orderBy(auctionBids.amount.desc())
      .limit(1);

    const minBid = highestBid ? Number(highestBid.amount) + 1 : 1; // add your real increment rule

    if (amount < minBid) {
      return res.status(400).json({
        error: `Bid must be at least ${minBid}`,
        currentHighest: highestBid ? highestBid.amount : null,
      });
    }

    // Insert bid
    const [bid] = await db
      .insert(auctionBids)
      .values({
        auctionId: auction.id,
        investorId: investor.id,
        amount,
      })
      .returning();

    // Check reserve
    const reserveMet = amount >= Number(auction.reservePrice);

    if (reserveMet) {
      // Immediately close auction and set winning bid
      await db
        .update(auctions)
        .set({
          status: "COMPLETED",
          winningBidId: bid.id,
        })
        .where(eq(auctions.id, auction.id));

      // Optionally: also update the bond status to "SOLD"
      // await updateBondStatusAfterSale(auction.bondId, bid.id);

      return res.json({
        bid,
        message: "Reserve met. Auction completed. You are the winning bidder.",
      });
    }

    return res.json({ bid, message: "Bid placed successfully" });
  }
);

// jobs/closeExpiredAuctions.ts
import { db } from "../db";
import { auctions } from "../db/schema/auctions";
import { and, eq, lt } from "drizzle-orm";

export async function closeExpiredAuctionsJob() {
  const now = new Date();

  const expired = await db
    .select()
    .from(auctions)
    .where(
      and(
        eq(auctions.status, "ACTIVE"),
        lt(auctions.endsAt, now)
      )
    );

  for (const a of expired) {
    // No reserve met if we’re still ACTIVE after end
    await db
      .update(auctions)
      .set({ status: "EXPIRED" })
      .where(eq(auctions.id, a.id));

    // Optionally: mark bond as "NOT SOLD" or return it to inventory
  }
}

General indemnity agreements accompany most surety bond contracts. This guide explains exactly how surety indemnity agreements work and what you are liable for when you sign one.

What Is an Indemnity Agreement?
A simple definition of indemnity is coverage for potential losses or damages. It means one party is financially liable if they cause loss to another. In the surety bond context, an indemnity agreement holds the bondholder liable for compensating the surety.

Indemnitors agree to hold harmless and indemnify the surety from financial loss caused by failure to perform of the principal. This transfers risk from the surety to the principal and is typically a requirement before the surety will issue a bond.

How Do Surety Indemnity Agreements Work?
In surety indemnity agreements, underwriters determine if the principal has the financial capacity to meet the contract terms. An indemnity bond works similarly to a line of credit by guaranteeing money is available. Like a bank, the surety expects repayment from the principal if it covers a claim. 

Business owners often establish an LLC to reduce personal liability for business finances. However, sureties require the personal indemnity of business owners to guarantee their obligation to repay the surety — even if the business goes bankrupt or is insolvent. Types of General Indemnity Agreement Provisions 
A principal may have different obligations in a general indemnity agreement (GIA) depending on the form’s specific terms. Here are the most common provisions found in surety bond indemnity agreements: 

Indemnification: A general indemnity provision transfers risk from the surety to the principal and covers a broad range of potential liabilities
Right to settle: The surety may act on the principal’s behalf and has full discretion on how to respond to a claim: pay, settle or defend
Right to enforce: The principal must fully cover any claims-related expenses of the surety, including legal fees
Books and records: The surety may inspect the principal’s books and financial records to ensure financial capacity 
Duty to cooperate: The principal must participate in any claims investigations
Collateral deposit: The surety has the right to request collateral security from indemnitors. (Only required for 0.1% of SuretyBonds.com customers.)
Who Signs the Indemnity Agreement?
Every stakeholder with 10% or more ownership of a business must sign the general indemnity agreement form individually and on behalf of the company. 

This way, if the principal is unable to repay a claim, the surety can collect payment from the individual owners. This personal indemnity makes it critical to avoid bond claims by acting ethically and following all laws and regulations. 

Why Does My Spouse Need to Sign? 
The spouses of any married business owners will also need to sign the surety indemnity agreement. Spousal indemnity protects the surety if business owners try to transfer assets into their spouses’ names to avoid repaying the surety for a claim. In some cases, a notary public must sign and notarize the GIA as well. 

Do All Bonds Require an Indemnity Agreement? 
Not necessarily. For some low-risk bonds, the surety may instantly issue the bond without a credit check or indemnity agreement. However, every application that requires underwriting will also need a signed indemnity form before the surety will issue the bond. 

Indemnity bonds have a higher risk of claim activity than other surety bonds due to the industry, certain terms and conditions, coverage amount — or a combination of factors. 

What Are Indemnity Bonds?
Indemnity bonds are a type of surety bond that creates a three-party contract between a business or individual, a surety provider, and an entity requiring the bond (obligee).

They function like any surety bond, but also require an indemnity agreement between the surety and the bondholder. Surety Bond Collateral Requirements

Many people may be surprised to find that collateral can become a vital part of the surety bond process. Bonding companies often require collateral they can secure against the provided services. Why Is Collateral Needed On Surety Bonds?
Collateral isn’t used too often, but it may be required in certain circumstances. A common misconception is that collateral is only needed for individuals or firms in financial distress. One of the reasons collateral may be required may be that bond types are at high risk to receive claims against their bond. For instance, defendants’ court bonds or appellate bonds, tax lien bonds, or release of lien bonds are a few types of bonds with a high claims frequency, so it makes good business sense for the surety to collect the money in advance. Collateral is used to support the applicant’s indemnity or agreement to repay the surety if the surety has to pay a claim. As a surety bond isn’t insurance, the applicant/principal must pay the surety back.

Poor credit is another, more obvious reason why someone would need to post collateral, but another less obvious reason is for someone with good credit to post collateral. Some applicants may have good credit and a lot of experience but do not have the financial strength in relation to the size of the bond, requiring them to supply collateral. In many cases, collateral may be needed because it’s a beneficial business decision made by the surety, so they know they will get paid back if there are any claims.

How Much Will I Have To Pay In Collateral?
The amount of the collateral depends on the surety, and the amount fluctuates with each instance. Therefore, the bond amount is a significant factor in determining the amount of collateral required.

Why Do I Have To Pay A Fee AND Pay Collateral?
Upon purchasing a surety bond, the fee required is only a small percentage of the bond, and the amount doesn’t go towards the collateral. Instead, the premium paid to purchase the bond is the sale of the bond itself towards services, etc.

How Long Is Collateral Held?
A surety bond company often holds collateral past the cancellation or release date of the bond because the obligee can still claim the bond up to a year or more after the bond is canceled. The collateral can be held up to 180 days but typically is returned within 90 days of the cancellation or release date of the bond.

Why Should I Get A Surety Bond Instead Of Just Posting An Irrevocable Letter Of Credit?
When there is an option to use an ILOC instead of getting a surety bond, why wouldn’t someone do that instead? The reason is quite simple. If only an ILOC is provided to the obligee without the placement of a surety bond, anyone can make a claim against you, and you will be required to pay it. With a surety bond, the surety performs research to determine whether or not the claim is valid.

What Is An Acceptable Form Of Collateral?
Generally, the only forms of collateral accepted are cash and an irrevocable letter of credit. An irrevocable letter of credit (ILOC) is a written guarantee by a financial institution that the necessary funds are available and cannot be touched for as long as the bond is active. For the entire period that the surety bond is active, the ILOC cannot be canceled or modified in any way unless there is an explicit agreement made by all parties involved. In brief, an ILOC serves as a promise that the bank will pay the surety if needed.

What Is Not An Acceptable Form Of Collateral?
Collateral cannot be a certificate of deposit (CD), government securities, or physical assets. Certificates of deposit are typically not accepted because their maturity dates rarely coincide with the bond term. The mismatched dates cause the collateral to liquidate prematurely, which will cause penalties or the collateral being held longer than it is needed.A CD is provided by a financial institution and is backed by the Federal Deposit Insurance Corporation. The CD allows the bank to retain the money for a certain amount of time, during which time the bank gains interest on the return, while they guarantee repayment once the term is complete. Government securities include blue-chip stocks and bonds and gold and silver bullion. These were accepted in the past for collateral, but since the market fluctuates, they are considered unreliable security for sureties. Physical assets are also not considered to be reliable collateral. What are Trusts?
A trust is a legal, fiduciary arrangement where assets are held by a third party for the benefit of other parties known as the beneficiary or beneficiaries. Trusts can be an important consideration in proper succession planning and estate planning. Trusts can hold many different types of assets including cash, marketable securities, life insurance, real property, etc.

There are also many different types of trusts. These can include marital trusts, generation skipping trusts, charitable trusts, life insurance trusts, etc. Two broad categories of trusts include Revocable Trusts and Irrevocable Trusts.

A Revocable Trust is one that can be changed during your lifetime. It allows adjustments to be made during one’s lifetime. A Revocable Trust can have some protection against creditors.

An Irrevocable Trust is the opposite. Once established, the rules are set. The benefit of this type of trust is that it often provides strong protection against creditors.

Trusts in Surety Bond Underwriting
Trusts can be a benefit in underwriting surety bonds. Trusts can provide a way to avoid probate and even potentially hold the life insurance or assets for the continuity of the company or the transition of that company. Trust can also potentially hinder surety bond underwriters, however. Here are some key underwriting considerations on how trusts can affect surety bonding.

Indemnity and Trusts
One of the key elements for extending surety bond credit is the Principle of Indemnity. That means that the surety bond company expects to be reimbursed by the Principal and indemnitors if they pay a valid loss. However, if a majority of a Principal’s assets are held in a trust, it may be difficult or impossible for the surety bond company to get reimbursed from that indemnitor. Therefore, one of the key considerations for providing surety bonds for an indemnitor with a trust is whether that trust can indemnify. This depends on how the trust was set up. If you have a trust, you should expect the surety bond underwriter to request a copy of the trust documents to see whether indemnity is available.

Personal indemnity is often required by surety bond companies when a person and/or their spouse owns 15% or more of a company. There are exceptions. Companies that have very strong balance sheets and track records, may seek a waiver of personal indemnity but in most cases, it will be required. When personal indemnity exists, expect the surety bond company to require the indemnity of that stockholder’s trust along with any trust owned by a spouse. This often brings up objections as it often contradicts the reason the trust was set up. Keep in mind that surety bonds are a credit product though. If a surety bond company cannot get the indemnity of the trust, they also cannot count those assets and the Principal may not qualify for surety bonding. Worse still, an important part of surety bond underwriting is Character. It is part of the 3 C’s which you can read about here. Part of character underwriting is the belief that a Principal will reimburse the surety bond company if a loss occurs. The unwillingness to include one’s trust is often a signal to the surety bond company that the Principal will not try to reimburse after a claim and may try to instead shelter their assets. This might cause some surety bond companies to decline an account, even if other aspects look strong.

Consider Limited Indemnity of the Trust
If your trust is strong enough, consider limiting the indemnity to a certain dollar figure to support your surety bond capacity. For example, say your trust holds $3 million in assets but your surety bond program only needs $1 million. Your surety bond underwriter may consider limiting the trust’s maximum exposure to $1 million of indemnity.

No Indemnity of the Trust
What if your trust is contractually not allowed to sign the indemnity agreement? Many times, the way a trust is structured, they are not allowed to sign these documents. In these cases, the surety bond company will have to rely only on the assets outside of the trust when deciding whether to extend surety bond credit.

Consider Your Lending Relationship
Surety Bond companies always want to be on the same level of indemnity as a Principal’s bank or lender. This means that if your trust is included on your other loans, the surety bond company will want it included in their indemnity as well. At times, a Principal will ask that the trust be excluded from the surety bond company’s indemnity, but this is difficult if the trust is being used as collateral for other credit purposes.

Leave Adequate Resources in the Company
The best strategy for many surety bond users is to hold adequate financial assets within the company to qualify for surety bond credit outside of your personal trust. By maintaining sufficient working capital and net worth in the company, personal assets, including trusts may be excluded.

// middleware/auth.ts
import { Request, Response, NextFunction } from "express";

export type AppRole = "ADMIN" | "UNDERWRITER" | "INTERNAL" | "INVESTOR";

export interface AuthUser {
  id: string;
  role: AppRole;
  email: string;
}

declare global {
  namespace Express {
    interface Request {
      user?: AuthUser;
    }
  }
}

export function requireAuth(
  allowedRoles?: AppRole[]
) {
  return (req: Request, res: Response, next: NextFunction) => {
    const user = req.user;
    if (!user) return res.status(401).json({ error: "Unauthorized" });

    if (allowedRoles && !allowedRoles.includes(user.role)) {
      return res.status(403).json({ error: "Forbidden" });
    }
    next();
  };
}

3. Backend Logic: Triggers, Auctions, Bidding, and Reserve
3.1. Automatic trigger: perfected bond → auction
Some other part of your app will mark bonds as perfected. When that happens – and the bond is approved for sale – we can auto-create an auction.

Event hook / service:

// services/bondPerfection.ts
import { db } from "../db";
import { bonds } from "../db/schema/bonds";
import { auctions } from "../db/schema/auctions";
import { eq } from "drizzle-orm";
import dayjs from "dayjs";

export async function handleBondPerfected(bondId: string, reservePrice: number) {
  const [bond] = await db.select().from(bonds).where(eq(bonds.id, bondId));
  if (!bond) throw new Error("Bond not found");

  const startsAt = new Date(); // or schedule/future start if needed
  const endsAt = dayjs(startsAt).add(5, "day").toDate();

  const [auction] = await db
    .insert(auctions)
    .values({
      bondId,
      reservePrice,
      status: "ACTIVE",
      startsAt,
      endsAt,
    })
    .returning();

  await db
    .update(bonds)
    .set({
      isPerfected: true,
      perfectedAt: new Date(),
      currentAuctionId: auction.id,
    })
    .where(eq(bonds.id, bondId));

  return auction;
}

investorBidsRouter.post(
  "/investor/auctions/:id/bids",
  requireAuth(["INVESTOR"]),
  async (req, res) => {
    const investor = req.user!;
    const { id } = req.params;
    const { amount } = req.body as { amount: number };

    if (!amount || amount <= 0) {
      return res.status(400).json({ error: "Bid amount must be positive" });
    }

    const [auction] = await db
      .select()
      .from(auctions)
      .where(eq(auctions.id, id));

    if (!auction) return res.status(404).json({ error: "Auction not found" });

    const now = new Date();
    if (auction.status !== "ACTIVE" || now < auction.startsAt || now > auction.endsAt) {
      return res.status(400).json({ error: "Auction is not open for bidding" });
    }

    // Get current highest bid
    const [highestBid] = await db
      .select()
      .from(auctionBids)
      .where(eq(auctionBids.auctionId, auction.id))
      .orderBy(auctionBids.amount.desc())
      .limit(1);

    const minBid = highestBid ? Number(highestBid.amount) + 1 : 1; // add your real increment rule

    if (amount < minBid) {
      return res.status(400).json({
        error: Bid must be at least ${minBid},
        currentHighest: highestBid ? highestBid.amount : null,
      });
    }

    // Insert bid
    const [bid] = await db
      .insert(auctionBids)
      .values({
        auctionId: auction.id,
        investorId: investor.id,
        amount,
      })
      .returning();

    // Check reserve
    const reserveMet = amount >= Number(auction.reservePrice);

    if (reserveMet) {
      // Immediately close auction and set winning bid
      await db
        .update(auctions)
        .set({
          status: "COMPLETED",
          winningBidId: bid.id,
        })
        .where(eq(auctions.id, auction.id));

      // Optionally: also update the bond status to "SOLD"
      // await updateBondStatusAfterSale(auction.bondId, bid.id);

      return res.json({
        bid,
        message: "Reserve met. Auction completed. You are the winning bidder.",
      });
    }

    return res.json({ bid, message: "Bid placed successfully" });

What Is an Indemnity Agreement?
A simple definition of indemnity is coverage for potential losses or damages. It means one party is financially liable if they cause loss to another. In the surety bond context, an indemnity agreement holds the bondholder liable for compensating the surety.

Indemnitors agree to hold harmless and indemnify the surety from financial loss caused by failure to perform of the principal. This transfers risk from the surety to the principal and is typically a requirement before the surety will issue a bond.

How Do Surety Indemnity Agreements Work?
In surety indemnity agreements, underwriters determine if the principal has the financial capacity to meet the contract terms. An indemnity bond works similarly to a line of credit by guaranteeing money is available. Like a bank, the surety expects repayment from the principal if it covers a claim. 

Business owners often establish an LLC to reduce personal liability for business finances. However, sureties require the personal indemnity of business owners to guarantee their obligation to repay the surety — even if the business goes bankrupt or is insolvent. Types of General Indemnity Agreement Provisions 
A principal may have different obligations in a general indemnity agreement (GIA) depending on the form’s specific terms. Here are the most common provisions found in surety bond indemnity agreements: 

Indemnification: A general indemnity provision transfers risk from the surety to the principal and covers a broad range of potential liabilities
Right to settle: The surety may act on the principal’s behalf and has full discretion on how to respond to a claim: pay, settle or defend
Right to enforce: The principal must fully cover any claims-related expenses of the surety, including legal fees
Books and records: The surety may inspect the principal’s books and financial records to ensure financial capacity 
Duty to cooperate: The principal must participate in any claims investigations
Collateral deposit: The surety has the right to request collateral security from indemnitors. (Only required for 0.1% of SuretyBonds.com customers.)
Who Signs the Indemnity Agreement?
Every stakeholder with 10% or more ownership of a business must sign the general indemnity agreement form individually and on behalf of the company. 

This way, if the principal is unable to repay a claim, the surety can collect payment from the individual owners. This personal indemnity makes it critical to avoid bond claims by acting ethically and following all laws and regulations. 

Why Does My Spouse Need to Sign? 
The spouses of any married business owners will also need to sign the surety indemnity agreement. Spousal indemnity protects the surety if business owners try to transfer assets into their spouses’ names to avoid repaying the surety for a claim. In some cases, a notary public must sign and notarize the GIA as well. 

Do All Bonds Require an Indemnity Agreement? 
Not necessarily. For some low-risk bonds, the surety may instantly issue the bond without a credit check or indemnity agreement. However, every application that requires underwriting will also need a signed indemnity form before the surety will issue the bond. 

Indemnity bonds have a higher risk of claim activity than other surety bonds due to the industry, certain terms and conditions, coverage amount — or a combination of factors.

Trusts can be a benefit in underwriting surety bonds. Indemnity and Trusts
One of the key elements for extending surety bond credit is the Principle of Indemnity. That means that the surety bond company expects to be reimbursed by the Principal and indemnitors if they pay a valid loss. Consider Limited Indemnity of the Trust
If your trust is strong enough, consider limiting the indemnity to a certain dollar figure to support your surety bond capacity. No Indemnity of the Trust

Consider Your Lending Relationship
Surety Bond companies always want to be on the same level of indemnity as a Principal’s bank or lender. This means that if your trust is included on your other loans, the surety bond company will want it included in their indemnity as well. 4. Underwriting Logic: How the App “Does the Surety Underwriting”
Your Surety engine should implement the same major duties a professional surety underwriter follows:

Due diligence on the principal & bond

Protecting trust assets / capacity control

Ensuring indemnity + collateral are in place

We’ll focus this on Judgment Bonds.

4.1. Checks on the trust before issuing surety
For each case:

Confirm trust authority

surety_trust.canActAsSurety === true

allowedBondTypes includes JUDGMENT.

Trust agreement on file includes authority to guarantee / pledge assets (you can track “trust docs verified” as a boolean).

Capacity limits

bondPenalSum <= maxSingleBondLimit.

(currentExposure + bondPenalSum) <= aggregateBondLimit.

Risk vs. insurance rule-of-thumb
From your doc: larger or higher-risk bonds warrant insurance or additional protections. 
Example rules:

If bondPenalSum <= 50,000 and fully collateralized ⇒ no extra insurance.

If 50,000 < bondPenalSum <= 250,000 ⇒ either insurance OR full collateral.

If > 250,000 ⇒ strong recommendation or requirement for insurance unless 100% collateralized.

4.2. Checks on the principal & owners (Judgment Bond-specific)
Evaluate principal’s financial capacity:

Net worth vs bond penal sum.

Cash flow / liquidity.

Existing judgments / defaults.

For owners (≥10–15%):

Each owner becomes an indemnitor and must sign.

Their spouses also must sign to prevent asset shielding transfers. 

Risk flags:

Prior bond claims.

Tax liens.

Bankruptcy history.

Very thin capital vs. huge judgment amount.

You can encode these into a simple numeric risk score (0–100) stored on surety_underwriting_cases.principalRiskScore.

4.3. Collateral decision rules
From your material:

Collateral is used when risk is high or bond types have high claim frequency.

Acceptable forms: cash or ILOC.

CDs, government securities, physical assets are generally not acceptable; real estate can be considered via separate process if you ever want to extend. 

Example algorithm for Judgment Bonds:

if principalRiskScore <= 30 and bondPenalSum <= 25,000:
    collateralRequiredAmount = 0
elif principalRiskScore <= 50 and bondPenalSum <= 100,000:
    collateralRequiredAmount = 0.25 * bondPenalSum
else:
    collateralRequiredAmount = 1.0 * bondPenalSum  (100% collateral)
And always limit collateral types to:

["CASH", "ILOC"]
The result is stored on:

surety_underwriting_cases.collateralRequiredAmount

surety_collateral record(s).

2. How this avoids data loss
We never DROP or recreate bonds.

We only ALTER TABLE bonds ADD COLUMN IF NOT EXISTS …
→ existing rows keep all their data exactly as-is.

New columns (e.g., is_perfected, surety_status) just get defaults.

All new tables (auctions, surety_*) are additive; they don’t touch existing rows.

If you know your bonds table already has bond_type or face_value columns with different types/names:

You can safely delete or comment out those specific ALTER TABLE … ADD COLUMN lines and:

Adjust the Drizzle schema in the zip to match your real column names/types or

Map them via a view or by renaming columns.

  return (req: Request, res: Response, next: NextFunction) => {

    if (!user) return res.status(401).json({ error: "Unauthorized" });

    if (allowedRoles && !allowedRoles.includes(user.role)) {

  generateDocsForBondStep,
  type BondProcessStep,
} from "@/lib/documentEngine";

// Optional: tighten allowed steps at runtime:
const ALLOWED_STEPS: BondProcessStep[] = [
  "JUDGMENT_TO_BOND",
  "PERFECTION",
  "SURETY_UNDERWRITING",
  "SURETY_ACTIVATION",
  "EXCHANGE_LISTING",
  "EXCHANGE_SALE",
  "TRUST_ALLOCATION",
];

export async function POST(
  req: NextRequest,
  context: { params: { bondId: string } },
) {
  try:
    // 🔐 Admin-only: throws Response if not authorized
    await requireAdminUser(req);

    const bondId = context.params.bondId;

      step: BondProcessStep | string;
      suretyCaseId?: string;

        { status: 400 },

1️⃣ documentEngine.ts
Put this somewhere like: src/server/documentEngine.ts (or lib/documentEngine.ts if that matches your layout).

// src/server/documentEngine.ts
import path from "path";
import fs from "fs/promises";
import Handlebars from "handlebars";
import { db } from "@/server/db"; // adjust to your db import
import { suretyDocuments } from "@/server/db/schema/suretyDocuments"; // adjust path

// All doc types we’re supporting right now
export type DocType =
  | "GIA"
  | "COLLATERAL_AGREEMENT"
  | "SURETY_TRUST_RESOLUTION"
  | "TRUST_ALLOCATION_RESOLUTION";

// Map docType → template filename
const TEMPLATE_MAP: Record<DocType, string> = {
  GIA: "gia.hbs",
  COLLATERAL_AGREEMENT: "collateral_agreement.hbs",
  SURETY_TRUST_RESOLUTION: "surety_trust_resolution.hbs",
  TRUST_ALLOCATION_RESOLUTION: "trust_allocation_resolution.hbs",
};

// Cache compiled templates so we don’t re-read on every call
const compiledTemplates: Partial<
  Record<DocType, Handlebars.TemplateDelegate>
> = {};

let helpersRegistered = false;

function ensureHelpersRegistered() {
  if (helpersRegistered) return;

  helpersRegistered = true;

  // === Shared helpers – must match what your templates use ===
  Handlebars.registerHelper("formatCurrency", function (value: any) {
    if (value == null) return "";
    const num = Number(value);
    if (Number.isNaN(num)) return String(value);
    return num.toLocaleString("en-US", {
      style: "currency",
      currency: "USD",
      minimumFractionDigits: 2,
    });
  });

  Handlebars.registerHelper("formatDate", function (value: any) {
    if (!value) return "";
    const d = new Date(value);
    if (Number.isNaN(d.getTime())) return String(value);
    return d.toLocaleDateString("en-US", {
      year: "numeric",
      month: "long",
      day: "numeric",
    });
  });

  Handlebars.registerHelper(
    "join",
    function (this: any, arr: any[], separator: string) {
      if (!Array.isArray(arr)) return "";
      return arr.join(separator);
    }
  );

  Handlebars.registerHelper("eq", function (a: any, b: any) {
    return a === b;
  });
}

// Load + compile template from /templates once, cache thereafter
async function getCompiledTemplate(docType: DocType) {
  ensureHelpersRegistered();

  if (compiledTemplates[docType]) {
    return compiledTemplates[docType]!;
  }

  const filename = TEMPLATE_MAP[docType];
  const templatePath = path.join(process.cwd(), "templates", filename);
  const source = await fs.readFile(templatePath, "utf8");

  const compiled = Handlebars.compile(source);
  compiledTemplates[docType] = compiled;
  return compiled;
}

// -------- PDF service abstraction --------

export interface PdfOptions {
  docType: DocType;
  caseId?: string;
  bondId?: string;
}

export interface PdfService {
  renderToPdf(html: string, options: PdfOptions): Promise<Buffer>;
}

/**
 * Default stub implementation: just returns the HTML as a Buffer.
 * Replace this with Puppeteer, Playwright, external API, etc.
 */
const defaultPdfService: PdfService = {
  async renderToPdf(html: string): Promise<Buffer> {
    // TODO: swap this out for a real PDF implementation
    return Buffer.from(html, "utf8");
  },
};

// -------- Public API: render + persist --------

export interface RenderAndPersistOptions {
  docType: DocType;
  context: any;          // BondDocContext or a subset; Handlebars will use it
  caseId: string;        // surety_cases.id
  pdfService?: PdfService;
}

/**
 * Renders a document from template + context, converts to PDF via pdfService,
 * writes it under /generated-docs, and inserts a row into surety_documents.
 *
 * Returns the DB record + storagePath.
 */
export async function renderDocumentAndPersist(
  opts: RenderAndPersistOptions
) {
  const { docType, context, caseId } = opts;
  const pdfService = opts.pdfService ?? defaultPdfService;

  // 1) Compile template
  const template = await getCompiledTemplate(docType);
  const html = template(context);

  // 2) Render HTML → PDF bytes
  const pdfBuffer = await pdfService.renderToPdf(html, {
    docType,
    caseId,
  });

  // 3) Decide where to store
  const filenameSafeType = docType.toLowerCase();
  const relPath = path.join(
    "surety",
    caseId,
    `${filenameSafeType}-${Date.now()}.pdf`
  );
  const absPath = path.join(process.cwd(), "generated-docs", relPath);

  await fs.mkdir(path.dirname(absPath), { recursive: true });
  await fs.writeFile(absPath, pdfBuffer);

  // 4) Insert into surety_documents
  const [docRecord] = await db
    .insert(suretyDocuments)
    .values({
      caseId,
      docType,
      storagePath: relPath,
      status: "DRAFT", // or "FULLY_EXECUTED" once signed, etc.
    })
    .returning();

  return {
    storagePath: relPath,
    document: docRecord,
  };
}
🔧 Where files live

Handlebars templates: process.cwd()/templates/*.hbs

Generated PDFs: process.cwd()/generated-docs/surety/{caseId}/...

DB insert: surety_documents table we designed earlier.

You can swap out defaultPdfService with Puppeteer / Playwright / an external HTML→PDF API very easily.

2️⃣ Example Next.js API usage
Here’s how you might call this from a Next.js App Router route like:

app/api/admin/surety/cases/[caseId]/documents/[docType]/route.ts

// app/api/admin/surety/cases/[caseId]/documents/[docType]/route.ts
import { NextRequest, NextResponse } from "next/server";
import {
  DocType,
  renderDocumentAndPersist,
} from "@/server/documentEngine";
import { getSuretyCaseContext } from "@/server/surety/getSuretyCaseContext"; // you implement this
import { requireAdminAuth } from "@/server/auth"; // your auth

type RouteParams = {
  params: {
    caseId: string;
    docType: DocType;
  };
};

export async function POST(req: NextRequest, { params }: RouteParams) {
  const user = await requireAdminAuth(req);

  const { caseId, docType } = params;

  // You can either:
  // 1) Build the full context server-side,
  // 2) Or accept some overrides in req.json().
  // Here we assume you have a helper that returns BondDocContext-like data.
  const baseContext = await getSuretyCaseContext(caseId);

  // Optional: body overrides (e.g. custom clause text)
  const body = await req.json().catch(() => ({}));
  const context = {
    ...baseContext,
    ...body?.overrides,
    meta: {
      ...(baseContext.meta ?? {}),
      generatedAt: new Date().toISOString(),
      generatedByUserId: user.id,
      docType,
    },
  };

  const result = await renderDocumentAndPersist({
    docType,
    context,
    caseId,
  });

  return NextResponse.json({
    ok: true,
    docType,
    storagePath: result.storagePath,
    document: result.document,
  });
}
You just need to implement getSuretyCaseContext(caseId) so it pulls the bond, principal, suretyTrust, collateral, trustAllocation, etc. from Drizzle and shapes it to what the templates expect.

// src/server/pdfService.ts
import { PdfService, PdfOptions } from "./documentEngine";
import { Browser } from "puppeteer"; // example, if you choose Puppeteer

export const puppeteerPdfService: PdfService = {
  async renderToPdf(html: string, options: PdfOptions): Promise<Buffer> {
    // Example with Puppeteer; fill in your actual implementation
    const browser: Browser = await /* get or create browser */;
    const page = await browser.newPage();
    await page.setContent(html, { waitUntil: "networkidle0" });
    const buffer = await page.pdf({ format: "A4" });
    await page.close();
    return buffer;
  },
};
