async function callGPTForExtraction(text: string): Promise<Partial<JudgmentExtractionResult>> {
  try {
    const completion = await openai.chat.completions.create({
      model: 'gpt-4o-mini',
      response_format: { type: 'json_object' },
      messages: [
        {
          role: 'system',
          content:
            'You extract structured data from court judgments. Respond as compact JSON with the keys: ' +
            'debtorName, caseNumber, court, jurisdiction, judgmentAmount, judgmentDate (ISO yyyy-mm-dd).'
        },
        { role: 'user', content: text }
      ]
    });

    const raw = completion.choices[0]?.message?.content ?? '{}';
    const parsed = JSON.parse(raw);

    const result: Partial<JudgmentExtractionResult> = {
      debtorName: parsed.debtorName,
      caseNumber: parsed.caseNumber,
      court: parsed.court,
      jurisdiction: parsed.jurisdiction,
      judgmentAmount: Number(parsed.judgmentAmount),
      judgmentDate: parsed.judgmentDate,
      rawFields: parsed
    };
    return result;
  } catch (err) {
    throw new ExternalServiceError('Failed to call OpenAI for extraction', err);
  }
}

/**
 * Dual-mode extractor: GPT + optional Claude.
 * If both available, merge and track confidence source per field.
 */
export async function extractJudgmentData(text: string): Promise<JudgmentExtractionResult> {
  const [gpt, claude] = await Promise.allSettled([
    callGPTForExtraction(text),
    callClaudeForExtraction(text)
  ]);

  const merged: any = {};
  const confidence: Record<string, ExtractionConfidenceSource> = {};

  const pickField = (field: string) => {
    const g = gpt.status === 'fulfilled' ? (gpt.value as any)[field] : undefined;
    const c = claude.status === 'fulfilled' ? (claude.value as any)[field] : undefined;

    if (g !== undefined && c !== undefined) {
      if (g === c) {
        merged[field] = g;
        confidence[field] = 'both';
      } else {
        // Heuristic: prefer Claude for legal text, GPT for numeric fields
        if (['debtorName', 'court', 'jurisdiction', 'caseNumber'].includes(field)) {
          merged[field] = c ?? g;
          confidence[field] = c !== undefined ? 'claude' : 'gpt';
        } else {
          merged[field] = g ?? c;
          confidence[field] = g !== undefined ? 'gpt' : 'claude';
        }
      }
    } else if (g !== undefined) {
      merged[field] = g;
      confidence[field] = 'gpt';
    } else if (c !== undefined) {
      merged[field] = c;
      confidence[field] = 'claude';
    }
  };

  pickField('debtorName');
  pickField('caseNumber');
  pickField('court');
  pickField('jurisdiction');
  pickField('judgmentAmount');
  pickField('judgmentDate');

  if (!merged.debtorName || !merged.caseNumber) {
    // As a fallback, return a minimal stub instead of throwing
    merged.debtorName = merged.debtorName ?? 'Unknown Debtor';
    merged.caseNumber = merged.caseNumber ?? 'UNKNOWN';
  }

  const rawFields =
    gpt.status === 'fulfilled'
      ? (gpt.value as any).rawFields ?? {}
      : claude.status === 'fulfilled'
      ? (claude.value as any).rawFields ?? {}
      : {};

  return {
    debtorName: merged.debtorName,
    caseNumber: merged.caseNumber,
    court: merged.court ?? 'Unknown Court',
    jurisdiction: merged.jurisdiction ?? 'Unknown Jurisdiction',
    judgmentAmount: Number(merged.judgmentAmount ?? 0),
    judgmentDate: merged.judgmentDate ?? new Date().toISOString().slice(0, 10),
    confidence,
    rawFields
  };
}

import { priceEstimates, judgments } from '../../db/schema.js';
import { db } from '../db/client.js';
import { eq } from 'drizzle-orm';
import { NotFoundError } from '../errors/index.js';

export interface PricingInput {
  judgmentId: number;
  targetIRR: number; // %
  discountRate: number; // %
  expectedRecoveryPct: number; // %
}

/**
 * Simple deterministic pricing model:
 * purchasePrice = judgmentAmount * (expectedRecoveryPct / 100) * (1 - discountRate / 100)
 * This can be refined later (Monte Carlo, term structure, etc.).
 */
export async function calculateAndPersistPrice(input: PricingInput) {
  const [judgment] = await db.select().from(judgments).where(eq(judgments.id, input.judgmentId));

  if (!judgment) {
    throw new NotFoundError('Judgment not found');
  }

  const expectedRecoveryAmount =
    Number(judgment.judgmentAmount) * (input.expectedRecoveryPct / 100);
  const purchasePrice =
    expectedRecoveryAmount * (1 - input.discountRate / 100);

  const [inserted] = await db
    .insert(priceEstimates)
    .values({
      judgmentId: input.judgmentId,
      targetIRR: input.targetIRR,
      discountRate: input.discountRate,
      recommendedPurchasePrice: purchasePrice,
      scenarioJson: {
        expectedRecoveryAmount,
        model: 'simple-linear',
        note:
          'Deterministic pricing. For production, replace with Monte Carlo or AI-augmented model.'
      }
    })
    .returning();

  return inserted;
}

export async function registerUser(email: string, password: string, role: string = 'analyst') {
  const [existing] = await db.select().from(users).where(eq(users.email, email));
  if (existing) {
    throw new ConflictError('Email already in use');
  }
  const passwordHash = await bcrypt.hash(password, 10);
  const [created] = await db
    .insert(users)
    .values({ email, passwordHash, role })
    .returning();
  return created;
}

export async function verifyUser(email: string, password: string) {
  const [user] = await db.select().from(users).where(eq(users.email, email));
  if (!user) {
    throw new AuthError('Invalid credentials');
  }
  const ok = await bcrypt.compare(password, user.passwordHash);
  if (!ok) {
    throw new AuthError('Invalid credentials');
  }
  return user;
}

export interface AuthUser {

  email: string;
  role: string;
}

declare module 'fastify' {
  interface FastifyRequest {
    user?: AuthUser;
  }
}

export function registerAuth(fastify: FastifyInstance) {
  fastify.register(import('fastify-jwt'), {
    secret: fastify.config.JWT_SECRET
  } as any);
}

export async function authGuard(request: FastifyRequest, _reply: FastifyReply) {
  try {
    const authHeader = request.headers.authorization;
    if (!authHeader?.startsWith('Bearer ')) {
      throw new AuthError('Missing token');
    }
    const token = authHeader.slice('Bearer '.length);
    const decoded = await (request as any).jwtVerify<AuthUser>({ token });
    request.user = decoded;
  } catch (err) {
    throw new AuthError('Invalid or missing token', err);
  }
}

    const parse = registerSchema.safeParse(req.body);
    if (!parse.success) {
      throw new ValidationError('Invalid register payload', parse.error.format());

    const token = (app as any).jwt.sign({

    reply.send({

      user: { id: user.id, email: user.email, role: user.role }
    });
  });

  app.post('/api/auth/login', async (req, reply) => {
    const parse = loginSchema.safeParse(req.body);
    if (!parse.success) {
      throw new ValidationError('Invalid login payload', parse.error.format());
    }
    const { email, password } = parse.data;
    const user = await verifyUser(email, password);
    const token = (app as any).jwt.sign({
      id: user.id,
      email: user.email,
      role: user.role
    });
    reply.send({
      token,
      user: { id: user.id, email: user.email, role: user.role }
    });
  });

    const user = (req as any).user;
    if (!user) {

    // NOTE: for demo we store pdfBase64 as rawText; in prod, persist file to S3/GCS.
    const [inserted] = await db
      .insert(judgments)
      .values({
        ownerId: user.id,
        debtorName: 'TBD',
        caseNumber: 'TBD',
        court: 'TBD',
        jurisdiction: 'TBD',
        judgmentAmount: 0,
        judgmentDate: new Date(),
        rawText: `PDF_BASE64_LENGTH=${parse.data.pdfBase64.length}`
      })
      .returning();
    return inserted;
  });

  app.post('/api/judgments/:id/extract', async (req) => {
    const user = (req as any).user!;
    const id = Number((req.params as any).id);
    const parse = extractSchema.safeParse(req.body);
    if (!parse.success) {
      throw new ValidationError('Invalid extract payload', parse.error.format());
    }
    const [row] = await db.select().from(judgments).where(eq(judgments.id, id));
    if (!row || row.ownerId !== user.id) {
      throw new NotFoundError('Judgment not found');
    }

    const result = await extractJudgmentData(parse.data.rawText);
    const [updated] = await db
      .update(judgments)
      .set({
        debtorName: result.debtorName,
        caseNumber: result.caseNumber,
        court: result.court,
        jurisdiction: result.jurisdiction,
        judgmentAmount: result.judgmentAmount,
        extractionJson: result.rawFields,
        extractionConfidence: result.confidence,
        rawText: parse.data.rawText,
        updatedAt: new Date()
      })
      .where(eq(judgments.id, id))
      .returning();

    return updated;
  });
}

# Judgment Bond AI – Product Requirements (Master)

## 1. Purpose & Vision

Judgment Bond AI transforms raw court judgments into investable bond-like assets by:
- Ingesting PDFs
- Extracting structured case & debtor data with AI
- Predicting recovery likelihood
- Computing fair purchase prices
- Bundling high-scoring judgments into tranches
- Generating investor-ready analytics and materials

It is **not legal or investment advice**. It is a decision-support and workflow tool.

## 2. Personas

- **Admin (You / Ops Owner)**
  - Manages users, roles, feature flags
  - Reviews AI extraction and pricing configs
- **Analyst (Underwriter)**
  - Uploads judgments, reviews AI extraction
  - Adjusts assumptions (recovery %, discount, IRR)
  - Builds bundles & tranches
- **Investor**
  - Views curated bundles and analytics
  - Reviews projected IRR, risk flags, stress tests

## 3. Core User Flows

1. **Judgment Ingestion Flow**
   - Analyst logs in
   - Uploads PDF or pastes raw text
   - Triggers AI extraction (GPT + optional Claude)
   - Reviews and edits fields
   - Saves canonical judgment record

2. **Recovery Scoring & Pricing**
   - Analyst selects a judgment
   - Runs AI recovery scoring
   - Chooses/edits target IRR and discount rate
   - System calculates recommended purchase price

3. **Bundle & Tranche Construction**
   - Analyst selects multiple judgments
   - Assigns tranches (Senior / Mezz / Equity)
   - App computes portfolio metrics (value, IRR proxy, risk mix)
   - Saves bundle for investors

4. **Investor Analytics & Pitch**
   - Investor views bundle dashboard
   - Sees key metrics + risk breakdown
   - Generates investor “teaser” view / PDF

5. **AI Onboarder**
   - User opens chat widget
   - Asks “How should I price this judgment?” or similar questions
   - Assistant responds with explanations referencing the data model & pricing logic

## 4. Non-Functional Requirements

- **Performance**
  - P95 API latency < 500ms for non-AI endpoints
  - AI endpoints dominated by provider response times

- **Reliability**
  - Clear error codes and retry guidance for AI calls
  - Defensive DB transactions

- **Security**
  - JWT auth, role-based permissions
  - Input validation via zod
  - PII redacted from logs

- **Compliance**
  - OWASP Top 10 mitigations addressed in SECURITY.md
  - Clear disclaimers: not legal/financial advice

- **Scalability**
  - Drizzle + Postgres schema with indexes

# Feature PRD – Judgment Ingestion

## 1. Problem Statement

Judgment buyers and underwriters receive court judgments as PDFs or scans that are:
- Hard to parse
- Error-prone to retype
- Inconsistent in format

This slows underwriting and introduces data quality risk.

## 2. Goals

- Allow analysts to quickly onboard new judgments into the system.
- Use AI to extract structured fields, but keep the human-in-control for final edits.
- Store raw text + structured data for later re-scoring and pricing.

Success metrics:
- Time from PDF upload to usable record < 2 minutes.
- At least 80% of fields correctly extracted by AI in typical cases (to be refined in later versions).

## 3. User Stories

1. **As an analyst**, I can upload a judgment PDF so that the system can extract key data.
2. **As an analyst**, I can review and edit extracted fields before saving.
3. **As an admin**, I can see which judgments were AI-extracted and when.

## 4. UX Notes

- Entry point: “Judgments” → “Upload Judgment”
- Fields displayed in a 2-column responsive form:
  - Debtor Name
  - Case Number
  - Court
  - Jurisdiction
  - Judgment Amount
  - Judgment Date
- “AI Extract” button appears after upload or text paste.
- AI status indicator:
  - Idle → “Ready to Extract”
  - Running → spinner + “Extracting with AI…”
  - Done → green check, per-field confidence tooltip (GPT / Claude / Both).

Empty states:
- No judgments yet → “Upload your first judgment to begin underwriting.”

## 5. Data Model

Uses `judgments` table:

- `ownerId`: FK → users.id (analyst that owns this record)
- `debtorName`: string
- `caseNumber`: string
- `court`: string
- `jurisdiction`: string
- `judgmentAmount`: decimal
- `judgmentDate`: datetime
- `status`: enum-like string [new, extracted, scored, priced, bundled]
- `rawText`: full text extracted from PDF
- `extractionJson`: JSON blob of raw AI output
- `extractionConfidence`: JSON mapping field → { gpt | claude | both }

## 6. API Contract (High Level)

Detailed in `/server/openapi.yml`.

- `POST /api/judgments/upload`
  - Request: `{ pdfBase64: string }`
  - Response: `Judgment` record with placeholder fields

- `POST /api/judgments/{id}/extract`
  - Request: `{ rawText: string }`
  - Behavior:
    - Runs dual-mode AI extraction
    - Updates judgment row
  - Response: updated `Judgment` record

- `GET /api/judgments`
  - Lists judgments owned by current user.

- `GET /api/judgments/{id}`
  - Returns a single judgment.

## 7. Error Handling

- 400 `VALIDATION_ERROR`
  - Payload missing or malformed.
- 401 `AUTH_ERROR`
  - Missing/invalid token.
- 404 `NOT_FOUND`
  - Judgment not found or not owned by user.
- 502 `EXTERNAL_SERVICE_ERROR`
  - AI provider timed out or responded with error.

Errors use standard format:

```json
{
  "code": "VALIDATION_ERROR",
  "message": "Invalid upload payload",
  "details": { "field": "pdfBase64" },
  "requestId": "req-123"
}

9. Acceptance Criteria
Uploading a synthetic PDF or base64 payload creates a judgments row.

Calling extract on that judgment populates fields and sets status = 'extracted'.

UI shows AI extraction confidence per field.

Unauthorized users cannot access other users’ judgments.

All requests are validated via zod.

+import { FastifyReply, FastifyRequest } from 'fastify';
+import { rateLimits } from '../../config/rateLimits.js';
+import { RateLimitError } from '../errors/index.js';
+
+type RateKey = 'aiExtraction' | 'pricing' | 'chat';
+
+interface Bucket {
+  count: number;
+  windowStart: number;
+}
+
+const windowMs = 60_000;
+const buckets = new Map<string, Bucket>();
+
+function getLimit(key: RateKey): number {
+  switch (key) {
+    case 'aiExtraction':
+      return rateLimits.aiExtractionPerMinute;
+    case 'pricing':
+      return rateLimits.pricingPerMinute;
+    case 'chat':
+      return rateLimits.chatPerMinute;
+  }
+}
+
+function bucketIdFor(req: FastifyRequest, key: RateKey): string {
+  const user = (req as any).user;
+  const ip =
+    (req.headers['x-forwarded-for'] as string) ||
+    req.ip ||
+    (req.raw.socket.remoteAddress ?? 'unknown');
+  if (user?.id) {
+    return `${key}:user:${user.id}`;
+  }
+  return `${key}:ip:${ip}`;
+}
+
+export async function rateLimit(
+  req: FastifyRequest,
+  _reply: FastifyReply,
+  key: RateKey
+): Promise<void> {
+  const id = bucketIdFor(req, key);
+  const now = Date.now();
+  const limit = getLimit(key);
+
+  const bucket = buckets.get(id);
+  if (!bucket) {
+    buckets.set(id, { count: 1, windowStart: now });
+    return;
+  }
+
+  if (now - bucket.windowStart > windowMs) {
+    bucket.count = 1;
+    bucket.windowStart = now;
+    return;
+  }
+
+  bucket.count += 1;
+
+  if (bucket.count > limit) {
+    throw new RateLimitError(
+      `Rate limit exceeded for ${key} (${bucket.count}/${limit} in current window)`
+    );
+  }
+}

+## Problem
+
+Analysts and investors need a fast, consistent way to turn raw judgment amounts into scenario-based pricing
+that can be compared across deals and exported into investor materials.
+
+## Goals
+
+- Deterministic, explainable pricing model (no hidden AI magic for the actual dollar math).
+- Support base / optimistic / stressed scenarios.
+- Persist scenarios to the database so they can be re-used in investor dashboards and bundles.
+- Expose a simple API so other services (UI, teaser generator, external tools) can call the engine.
+
+## Flows
+
+1. Analyst uploads / extracts a judgment.
+2. Analyst clicks “Run Pricing” on the judgment:
+   - Frontend calls `POST /api/judgments/{id}/price` with target IRR, discount and expected recovery %.
+   - API runs `pricingService.priceJudgmentWithScenarios` which:
+     - Loads the judgment.
+     - Builds base / optimistic / stressed scenarios.
+     - Calculates recommended prices per scenario.
+     - Writes rows into `price_estimates` keyed by `scenario_name`.
+3. Frontend fetches scenarios from `GET /api/judgments/{id}/price-estimates` and renders a comparison table.
+
+## Data model
+
+- `price_estimates` table:
+  - `judgment_id` FK → `judgments`.
+  - `target_irr`, `discount_rate`, `recommended_purchase_price`.
+  - `scenario_name` enum-ish string: `base`, `optimistic`, `stressed`.
+  - `scenario_json` stores derived fields (expected recovery amount, notes, etc).
+
+## APIs
+
+- `POST /api/judgments/{id}/price`
+- `GET /api/judgments/{id}/price-estimates`
+
+## Errors & Edge Cases
+
+- Judgment not found → 404.
+- Invalid payload / negative numbers → 400.
+- Rate limited (`pricing` bucket) for abuse prevention.

+
+## Goals
+
+- Enable analysts to create bundles and assign judgments to Senior / Mezz / Equity tranches.
+- Use base-scenario pricing as the reference price for bundle metrics.
+- Provide a single “Bundle detail” view that includes overall and per-tranche stats.
+
+## Flows
+
+1. Analyst opens the Bundles screen.
+2. Analyst creates a new bundle.
+3. Analyst selects a bundle and uses the “Tranche Builder” panel to:
+   - Add judgments to a tranche.
+   - Move a judgment between tranches.
+   - Remove a judgment from the bundle.
+4. The API recomputes bundle metrics based on:
+   - Face value (sum of `judgment_amount`).
+   - Recommended price (sum of base-scenario `recommended_purchase_price`).
+5. Investor dashboards and teaser exports use the same bundle metrics.
+
+## Data model
+
+- `bundles` table (already exists).
+- `bundle_judgments` link table with `tranche` string column.
+- Aggregations are computed on the fly from:
+  - `judgments`
+  - `bundle_judgments`
+  - base-scenario rows in `price_estimates`.
+
+## APIs
+
+- `GET /api/bundles`
+- `POST /api/bundles`
+- `GET /api/bundles/{id}`
+- `PATCH /api/bundles/{id}`
+- `POST /api/bundles/{id}/judgments`
+- `DELETE /api/bundles/{id}/judgments/{judgmentId}`
+
+## Analytics
+
+- Bundles created per user.
+- Average face / price per tranche and overall.
